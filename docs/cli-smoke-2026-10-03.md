## Reproducible minimum

After the normal dependency setup, run from the repository root. The environment
wrapper below makes the isolation used here explicit. `AUDIT_ROOT` stands for a
new directory and must not point to existing user data.

```sh
npm run build
AUDIT_ROOT=$(mktemp -d /tmp/flyrewheel-cli-smoke-XXXXXX)
mkdir "$AUDIT_ROOT/home"
cli() {
  env -i PATH="$PATH" HOME="$AUDIT_ROOT/home" TMPDIR="$AUDIT_ROOT" \
    LANG=C.UTF-8 QE_ENABLE_MODEL=false QE_S3_ENABLED=false \
    node dist/cli.js "$@"
}

cli closed-loop demo --out-dir "$AUDIT_ROOT/loop" --db "$AUDIT_ROOT/loop-db"
cli closed-loop demo --out-dir "$AUDIT_ROOT/loop" --db "$AUDIT_ROOT/loop-db"
cli revisions comparisons --db "$AUDIT_ROOT/loop-db"
cli revisions decisions --db "$AUDIT_ROOT/loop-db"

cli rules import --rule examples/semantic-rule-v2.json \
  --cases examples/semantic-rule-cases.json --db "$AUDIT_ROOT/rules-db"
cli rules list --db "$AUDIT_ROOT/rules-db"
RULE_DIGEST=610c2b5535fecb3cecbdfc5ba4c55b3e00992264ec88357ac12130bd22c7e459
cli rules show --digest "$RULE_DIGEST" --db "$AUDIT_ROOT/rules-db"
cli rules import --rule examples/semantic-rule-v2.json \
  --cases examples/semantic-rule-cases.json --db "$AUDIT_ROOT/rules-db"

cli demo --db "$AUDIT_ROOT/demo-db" --out "$AUDIT_ROOT/demo-report.json"
```
