1222 |     dimension of the hidden_states tensor"""
1223 |     # Scorer hidden states. For prefill step, it is used for hidden states of
1224 |     # all tokens, whereas for decode step, it use used for last accepted tokens.
1225 |     hidden_states: torch.Tensor
1226 |     # The sequence group metadata list. Only needed for decode step.
1227 |     seq_group_metadata_list: Optional[List[SequenceGroupMetadata]] = None
1228 |     # Scorer hidden states of the 2nd last token proposed by the proposer (
1229 |     # irrespective of whether it was accepted or not). Only used for cases when
1230 |     # last proposed token is accepted (i.e., in case of bonus tokens). For the
1231 |     # case of no bonus tokens, these are ignored.
1232 |     second_last_token_hidden_states: Optional[torch.Tensor] = None

1234 |     _seq_ids: List[int] = msgspec.field(default_factory=list)

1236 |     def __post_init__(self):

1238 |             assert len(self.seq_group_metadata_list) == len(self.hidden_states)
1239 |             self._seq_ids = get_all_seq_ids(self.seq_group_metadata_list)
1240 |

1243 |         return self._seq_ids

1249 |         """Update hidden states from target model invocation. Only used for

1251 |         assert len(seq_group_metadata_list) == len(hidden_states)
1252 |         self._seq_ids.extend(get_all_seq_ids(seq_group_metadata_list))
1253 |         self.hidden_states = torch.cat([self.hidden_states, hidden_states])
1254 |
1255 |         if self.second_last_token_hidden_states is not None:
1256 |             # Adding dummy hidden_states to this to maintain same shape
1257 |             self.second_last_token_hidden_states = torch.cat([

1264 |     def prune(self,
1265 |               seq_group_metadata_list: List[SequenceGroupMetadata]) -> None:
1266 |         """Prune to provided list of sequence ids. Only used for decode steps.
1267 |         """
1268 |         # Currently this prunes all seq_ids not present in
1269 |         # seq_group_metadata_list which might cause problems where a sequence
1270 |         # may be "paused" then "resumed" later. This should only prune sequences
1271 |         # which are confirmed to be aborted.
1272 |         seq_ids = get_all_seq_ids(seq_group_metadata_list)
1273 |         if seq_ids != self._seq_ids:
1274 |             # Batch contents changed - prune removed sequences.
1275 |             index = [self._seq_ids.index(seq_id) for seq_id in seq_ids]
1276 |             self.hidden_states = self.hidden_states[index]
1277 |             if self.second_last_token_hidden_states is not None:
1278 |                 self.second_last_token_hidden_states = self\
1279 |                     .second_last_token_hidden_states[index]

1284 |         """Expand hidden states for sequences with bonus tokens. This is in
1285 |         alignment with `MultiStepWorker._expand_execute_model_request`."""
1286 |         if self.second_last_token_hidden_states is None \

1288 |             return
1289 |
1290 |         index = []
1291 |         for seq_id in self._seq_ids:

1294 |                 index.append(i + len(self._seq_ids))
1295 |             index.append(i)
