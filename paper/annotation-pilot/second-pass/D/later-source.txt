0006 | import torch

0008 | _SAMPLING_EPS = 1e-5

0011 | class SamplingType(IntEnum):
0012 |     GREEDY = 0

0014 |     BEAM = 2

0016 |
0017 | LogitsProcessor = Callable[[List[int], torch.Tensor], torch.Tensor]
0018 | """LogitsProcessor is a function that takes a list of previously generated
0019 | tokens and a tensor of the logits for the next token, and returns a modified
0020 | tensor of logits to sample from."""

0023 | class SamplingParams:
0024 |     """Sampling parameters for text generation.
0025 |
0026 |     Overall, we follow the sampling parameters from the OpenAI text completion
0027 |     API (https://platform.openai.com/docs/api-reference/completions/create).
0028 |     In addition, we support beam search, which is not supported by OpenAI.
0029 |
0030 |     Args:
0031 |         n: Number of output sequences to return for the given prompt.
0032 |         best_of: Number of output sequences that are generated from the prompt.
0033 |             From these `best_of` sequences, the top `n` sequences are returned.
0034 |             `best_of` must be greater than or equal to `n`. This is treated as
0035 |             the beam width when `use_beam_search` is True. By default, `best_of`
0036 |             is set to `n`.
0037 |         presence_penalty: Float that penalizes new tokens based on whether they
0038 |             appear in the generated text so far. Values > 0 encourage the model
0039 |             to use new tokens, while values < 0 encourage the model to repeat
0040 |             tokens.
0041 |         frequency_penalty: Float that penalizes new tokens based on their
0042 |             frequency in the generated text so far. Values > 0 encourage the
0043 |             model to use new tokens, while values < 0 encourage the model to
0044 |             repeat tokens.
0045 |         repetition_penalty: Float that penalizes new tokens based on whether
0046 |             they appear in the prompt and the generated text so far. Values > 1
0047 |             encourage the model to use new tokens, while values < 1 encourage
0048 |             the model to repeat tokens.
0049 |         temperature: Float that controls the randomness of the sampling. Lower
0050 |             values make the model more deterministic, while higher values make
0051 |             the model more random. Zero means greedy sampling.
0052 |         top_p: Float that controls the cumulative probability of the top tokens
0053 |             to consider. Must be in (0, 1]. Set to 1 to consider all tokens.
0054 |         top_k: Integer that controls the number of top tokens to consider. Set
0055 |             to -1 to consider all tokens.
0056 |         min_p: Float that represents the minimum probability for a token to be
0057 |             considered, relative to the probability of the most likely token.
0058 |             Must be in [0, 1]. Set to 0 to disable this.
0059 |         use_beam_search: Whether to use beam search instead of sampling.
0060 |         length_penalty: Float that penalizes sequences based on their length.
0061 |             Used in beam search.
0062 |         early_stopping: Controls the stopping condition for beam search. It
0063 |             accepts the following values: `True`, where the generation stops as
0064 |             soon as there are `best_of` complete candidates; `False`, where an
0065 |             heuristic is applied and the generation stops when is it very
0066 |             unlikely to find better candidates; `"never"`, where the beam search
0067 |             procedure only stops when there cannot be better candidates
0068 |             (canonical beam search algorithm).
0069 |         stop: List of strings that stop the generation when they are generated.
0070 |             The returned output will not contain the stop strings.
0071 |         stop_token_ids: List of tokens that stop the generation when they are
0072 |             generated. The returned output will contain the stop tokens unless
0073 |             the stop tokens are special tokens.
0074 |         include_stop_str_in_output: Whether to include the stop strings in output
0075 |             text. Defaults to False.
0076 |         ignore_eos: Whether to ignore the EOS token and continue generating
0077 |             tokens after the EOS token is generated.
0078 |         max_tokens: Maximum number of tokens to generate per output sequence.
0079 |         logprobs: Number of log probabilities to return per output token.
0080 |             Note that the implementation follows the OpenAI API: The return
0081 |             result includes the log probabilities on the `logprobs` most likely
0082 |             tokens, as well the chosen tokens. The API will always return the
0083 |             log probability of the sampled token, so there  may be up to
0084 |             `logprobs+1` elements in the response.
0085 |         prompt_logprobs: Number of log probabilities to return per prompt token.
0086 |         skip_special_tokens: Whether to skip special tokens in the output.
0087 |         spaces_between_special_tokens: Whether to add spaces between special
0088 |             tokens in the output.  Defaults to True.
0089 |         logits_processors: List of functions that modify logits based on
0090 |             previously generated tokens.
0091 |     """

0160 |     def _verify_args(self) -> None:

0165 |                              f"got n={self.n} and best_of={self.best_of}.")

0167 |             raise ValueError("presence_penalty must be in [-2, 2], got "
0168 |                              f"{self.presence_penalty}.")
0169 |         if not -2.0 <= self.frequency_penalty <= 2.0:
0170 |             raise ValueError("frequency_penalty must be in [-2, 2], got "
0171 |                              f"{self.frequency_penalty}.")
0172 |         if not 0.0 < self.repetition_penalty <= 2.0:
0173 |             raise ValueError("repetition_penalty must be in (0, 2], got "
0174 |                              f"{self.repetition_penalty}.")
0175 |         if self.temperature < 0.0:
0176 |             raise ValueError(
0177 |                 f"temperature must be non-negative, got {self.temperature}.")
0178 |         if not 0.0 < self.top_p <= 1.0:

0196 |     def _verify_beam_search(self) -> None:
0197 |         if self.best_of == 1:
0198 |             raise ValueError("best_of must be greater than 1 when using beam "
0199 |                              f"search. Got {self.best_of}.")
0200 |         if self.temperature > _SAMPLING_EPS:
0201 |             raise ValueError("temperature must be 0 when using beam search.")
0202 |         if self.top_p < 1.0 - _SAMPLING_EPS:
0203 |             raise ValueError("top_p must be 1 when using beam search.")
0204 |         if self.top_k != -1:
0205 |             raise ValueError("top_k must be -1 when using beam search.")
0206 |         if self.early_stopping not in [True, False, "never"]:
0207 |             raise ValueError(
0208 |                 f"early_stopping must be True, False, or 'never', "
0209 |                 f"got {self.early_stopping}.")
0210 |
0211 |     def _verify_non_beam_search(self) -> None:
0212 |         if self.early_stopping is not False:
0213 |             raise ValueError("early_stopping is not effective and must be "
0214 |                              "False when not using beam search.")
0215 |         if (self.length_penalty < 1.0 - _SAMPLING_EPS
0216 |                 or self.length_penalty > 1.0 + _SAMPLING_EPS):
0217 |             raise ValueError(
0218 |                 "length_penalty is not effective and must be the "
0219 |                 "default value of 1.0 when not using beam search.")

0221 |     def _verify_greedy_sampling(self) -> None:

0224 |                              f"Got {self.best_of}.")
